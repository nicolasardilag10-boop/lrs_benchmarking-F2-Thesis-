    58	# Read samples.tsv, verify its required columns, reject missing values, and
    59	# return the sample name, sequencing technology, and FASTQ path for each row
    60	def read_sample_sheet(path):
    61	    path = Path(path)
    62	
    63	    if not path.exists():
    64	        raise FileNotFoundError(
    65	            f"Missing sample sheet: {path}"
    66	        )
    67	
    68	    rows = []
    69	
    70	    with path.open(newline="") as handle:
    71	        reader = csv.DictReader(
    72	            handle,
    73	            delimiter="\t"
    74	        )
    75	
    76	        required_columns = {
    77	            "sample",
    78	            "technology",
    79	            "fastq",
    80	        }
    81	
    82	        observed_columns = set(
    83	            reader.fieldnames or []
    84	        )
    85	
    86	        missing_columns = (
    87	            required_columns - observed_columns
    88	        )
    89	
    90	        if missing_columns:
    91	            raise ValueError(
    92	                f"Sample sheet is missing columns: "
    93	                f"{sorted(missing_columns)}"
    94	            )
    95	
    96	        for row in reader:
    97	            sample = row["sample"].strip()
    98	            technology = row["technology"].strip()
    99	            fastq = row["fastq"].strip()
   100	
   101	            if not sample or not technology or not fastq:
   102	                raise ValueError(
   103	                    f"Invalid empty value in sample sheet row: {row}"
   104	                )
   105	
   106	            if technology not in {"ont", "pb"}:
   107	                raise ValueError(
   108	                    f"Unsupported sequencing technology: {technology}"
   109	                )
   110	
   111	            rows.append(
   112	                {
   113	                    "sample": sample,
   114	                    "technology": technology,
   115	                    "fastq": fastq,
   116	                }
   117	            )
   118	
   119	    if not rows:
   120	        raise ValueError(
   121	            f"Sample sheet is empty: {path}"
   122	        )
   123	
   124	    return rows

     1	#####################################################################################
     2	# INPUT AND ENVIRONMENT VALIDATION
     3	# This module checks the sample table, sequencing files, and Conda environment
     4	# definitions before any assembler or scaffolding rule is executed.
     5	#####################################################################################
     6	
     7	
     8	# -----------------------------------------------------------------------------
     9	# Rule 1: Validate workflow inputs
    10	# -----------------------------------------------------------------------------
    11	
    12	# Check the required sample table, FASTQ files, and environment definitions to
    13	# prevent the workflow from failing later because of an incomplete setup
    14	rule validate_inputs:
    15	    output:
    16	        VALIDATION_OK
    17	    run:
    18	        Path(output[0]).parent.mkdir(
    19	            parents=True,
    20	            exist_ok=True
    21	        )
    22	
    23	        errors = []
    24	
    25	        # Verify that samples.tsv exists
    26	        if not Path(SAMPLE_SHEET).exists():
    27	            errors.append(
    28	                f"Missing sample sheet: {SAMPLE_SHEET}"
    29	            )
    30	
    31	        # Verify that every FASTQ declared in samples.tsv exists
    32	        for row in SAMPLE_ROWS:
    33	            fastq = Path(row["fastq"])
    34	
    35	            if not fastq.exists():
    36	                errors.append(
    37	                    f"Missing FASTQ file: {fastq}"
    38	                )
    39	
    40	        # Build the list of required environments from active_assemblers
    41	        required_environments = []
    42	
    43	        if "flye" in ACTIVE_ASSEMBLERS:
    44	            required_environments.append(
    45	                ENV_FLYE
    46	            )
    47	
    48	        if "goldrush" in ACTIVE_ASSEMBLERS:
    49	            required_environments.append(
    50	                ENV_GOLDRUSH
    51	            )
    52	
    53	        if "ntlink" in ACTIVE_ASSEMBLERS:
    54	            required_environments.append(
    55	                ENV_NTLINK
    56	            )
    57	
    58	        if "verkko" in ACTIVE_ASSEMBLERS:
    59	            required_environments.append(
    60	                ENV_VERKKO
    61	            )
    62	
    63	        # Verify that every required environment YAML file exists
    64	        for environment_file in required_environments:
    65	            if not Path(environment_file).exists():
    66	                errors.append(
    67	                    f"Missing Conda environment file: "
    68	                    f"{environment_file}"
    69	                )
    70	
    71	        # Stop the workflow and display all detected problems together
    72	        if errors:
    73	            message = "\n".join(errors)
    74	
    75	            raise RuntimeError(
    76	                "\nASSEMBLER WORKFLOW VALIDATION FAILED\n"
    77	                "Fix these problems before running the workflow:\n\n"
    78	                f"{message}\n"
    79	            )
    80	
    81	        # Create the marker file after all validations pass
    82	        Path(output[0]).write_text(
    83	            "Assembler input validation passed.\n"
    84	        )

     1	###############################################################################
     2	# GOLDRUSH ASSEMBLY RULES
     3	###############################################################################
     4	
     5	rule goldrush_assembly:
     6	    input:
     7	        validation="results/logs/validate_inputs.ok",
     8	        reads=lambda wildcards: fastq_for(wildcards)
     9	    output:
    10	        assembly="results/smoke_test/goldrush/{sample}.{technology}/assembly.fasta"
    11	    params:
    12	        workdir=lambda wildcards: (
    13	            f"results/smoke_test/goldrush/"
    14	            f"{wildcards.sample}.{wildcards.technology}/work"
    15	        ),
    16	        prefix=lambda wildcards: (
    17	            f"{wildcards.sample}_{wildcards.technology}_goldrush"
    18	        ),
    19	        reads_basename=lambda wildcards: (
    20	            f"{wildcards.sample}_{wildcards.technology}"
    21	        ),
    22	        min_read_length=lambda wildcards: (
    23	            5000 if wildcards.technology == "ont" else 10000
    24	        )
    25	    threads:
    26	        lambda wildcards: int(config.get("threads", 4))
    27	    conda:
    28	        ENV_GOLDRUSH
    29	    shell:
    30	        r"""
    31	        set -euo pipefail
    32	
    33	        mkdir -p "{params.workdir}"
    34	        mkdir -p "$(dirname "{output.assembly}")"
    35	
    36	        # GoldRush expects an uncompressed FASTQ in its working directory.
    37	        gzip -cd "{input.reads}" \
    38	            > "{params.workdir}/{params.reads_basename}.fastq"
    39	
    40	        cd "{params.workdir}"
    41	
    42	        goldrush run \
    43	            reads="{params.reads_basename}" \
    44	            G={config[genome_size]} \
    45	            t={threads} \
    46	            m={params.min_read_length} \
    47	            P=0 \
    48	            p="{params.prefix}" \
    49	            track_time=1
    50	
    51	        FINAL_ASSEMBLY=$(
    52	            find goldrush_intermediate_files \
    53	                -type f \
    54	                -name "*ntLink-5rounds.polished.fa" \
    55	                -print \
    56	                | sort \
    57	                | tail -n 1
    58	        )
    59	
    60	        if [[ -z "$FINAL_ASSEMBLY" || ! -s "$FINAL_ASSEMBLY" ]]; then
    61	            echo "ERROR: GoldRush final polished assembly was not produced."
    62	            exit 1
    63	        fi
    64	
    65	        cp "$FINAL_ASSEMBLY" \
    66	           "$OLDPWD/{output.assembly}"
    67	
    68	        test -s "$OLDPWD/{output.assembly}"
    69	        """

     1	#####################################################################################
     2	# NTLINK SCAFFOLDING RULES
     3	# ntLink uses long reads to order, orient, and connect contigs from an existing
     4	# draft assembly generated by another assembler such as Flye or GoldRush.
     5	#####################################################################################
     6	
     7	
     8	# -----------------------------------------------------------------------------
     9	# Future Rule 4: ntLink scaffolding
    10	# -----------------------------------------------------------------------------
    11	
    12	# Planned inputs:
    13	#   - Input validation marker
    14	#   - Draft assembly FASTA from Flye or GoldRush
    15	#   - ONT or PacBio long-read FASTQ file
    16	#
    17	# Planned outputs:
    18	#   - Scaffolded assembly FASTA
    19	#   - ntLink log and intermediate files
    20	#
    21	# ntLink is not treated as an independent de novo assembler because it requires
    22	# a previously generated draft assembly.
    23	
    24	
    25	
    26	rule ntlink_scaffold:
    27	    input:
    28	        draft="results/smoke_test/goldrush/{sample}.{technology}/assembly.fasta",
    29	        reads=lambda wildcards: fastq_for(wildcards)
    30	
    31	    output:
    32	        assembly="results/smoke_test/ntlink/{sample}.{technology}/assembly.fasta"
    33	
    34	    threads: 4
    35	
    36	    conda:
    37	        ENV_NTLINK
    38	
    39	    params:
    40	        workdir="results/smoke_test/ntlink/{sample}.{technology}/work"
    41	
    42	    shell:   #start the bash command for the rule
    43	        #start a multiline section. Set -euo pipefail makes to stop de script if real error happens. mkdir create the specific nt Link working directory
    44	        r"""
    45	        set -euo pipefail
    46	
    47	        mkdir -p "{params.workdir}"
    48	        mkdir -p "$(dirname "{output.assembly}")"
    49	
    50	        cp "{input.draft}" "{params.workdir}/draft.fasta"
    51	        gzip -cd "{input.reads}" \
    52	            > "{params.workdir}/reads.fastq"
    53	
    54	        (
    55	            cd "{params.workdir}"
    56	
    57	            ntLink scaffold \
    58	                target=draft.fasta \
    59	                reads='reads.fastq' \
    60	                t={threads}
    61	        )
    62	
    63	        cp "{params.workdir}/draft.fasta.k32.w100.z1000.ntLink.scaffolds.fa" \
    64	           "{output.assembly}"
    65	
    66	        test -s "{output.assembly}"
    67	        """
